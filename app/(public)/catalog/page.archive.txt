/**
 * Catalog page — Server Component.
 * Passes DB categories to CatalogClient for the sidebar.
 * Products are rendered from static data in CatalogClient (no DB fetch needed).
 */
import type { Metadata } from 'next';
import { createClient } from '@/lib/supabase/server';
import type { Category } from '@/lib/types';
import CatalogClient from '@/components/catalog/CatalogClient';

export const metadata: Metadata = {
  title: 'B2B Catalog — Nobilitas Ventures',
  description:
    'Browse our full B2B catalog of cleaning, office, hospitality, and industrial supplies. ' +
    'Pricing on inquiry. Request a quote from Nobilitas Ventures, Jaipur.',
};

interface CatalogPageProps {
  searchParams: Promise<{ category?: string }>;
}

export default async function CatalogPage({ searchParams }: CatalogPageProps) {
  const params = await searchParams;
  const initialCategory = params.category?.trim() ?? '';

  // Fetch DB categories for the sidebar (may be empty if DB not populated)
  const supabase = await createClient();
  const { data } = await supabase
    .from('categories')
    .select('*')
    .eq('is_active', true)
    .order('display_order');

  const serverCategories: Category[] = data ?? [];

  return (
    <CatalogClient
      initialCategory={initialCategory}
      serverCategories={serverCategories}
    />
  );
}
